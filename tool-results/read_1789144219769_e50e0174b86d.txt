    33→/**
    34→ * NOC threat-condition level (DEFCON-style): derives posture from routes +
    35→ * active run phase. L5 all-clear → L1 active hijack quarantine.
    36→ */
    37→const VALID_TABS = ['control', 'analytics', 'benchmark', 'copilot', 'docs'] as const;
    38→const PREFIX_URL_RE = /^\d{1,3}(?:\.\d{1,3}){3}\/\d{1,2}$/;
    39→
    40→/** read the initial deep-link (?tab=…&prefix=…) once, before first paint of state */
    41→function readDeepLink(): { tab: string; prefix: string | null } {
    42→  if (typeof window === 'undefined') return { tab: 'control', prefix: null };
    43→  try {
    44→    const params = new URLSearchParams(window.location.search);
    45→    const tab = params.get('tab');
    46→    const prefix = params.get('prefix');
    47→    return {
    48→      tab: (VALID_TABS as readonly string[]).includes(tab ?? '') ? (tab as string) : 'control',
    49→      prefix: prefix && PREFIX_URL_RE.test(prefix) ? prefix : null,
    50→    };
    51→  } catch {
    52→    return { tab: 'control', prefix: null };
    53→  }
    54→}

    56→function ThreatCondition({ state }: { state: SimState }) {
    57→  const routes = Object.values(state.routes);
    58→  const quarantined = routes.filter((r) => r.underOverride && r.route.locPref === 0).length;
    59→  const anomalous = routes.filter((r) => r.status !== 'normal' || r.underOverride).length;
    60→  const phase = state.activeRun?.phase;
    61→
    62→  let level = 5;
    63→  let label = 'ALL CLEAR';
    64→  let tone = 'border-emerald-800 text-emerald-300 bg-emerald-950/40';
    65→  if (quarantined > 0 || phase === 'mitigated') {
    66→    level = 1;
    67→    label = 'QUARANTINE';
    68→    tone = 'border-red-700 text-red-300 bg-red-950/50';
    69→  } else if (phase === 'detected' || phase === 'injected') {
    70→    level = 2;
    71→    label = 'ENGAGED';
    72→    tone = 'border-amber-700 text-amber-300 bg-amber-950/40';
    73→  } else if (anomalous > 0) {
    74→    level = 3;
    75→    label = 'ELEVATED';
    76→    tone = 'border-orange-800 text-orange-300 bg-orange-950/40';
    77→  } else if (state.metrics.totalRuns > 0 && state.metrics.msrPercent < 100) {
    78→    level = 4;
    79→    label = 'GUARDED';
    80→    tone = 'border-yellow-800 text-yellow-300 bg-yellow-950/30';
    81→  }
    82→
    83→  return (
    84→    <span
    85→      className={`hidden sm:inline-flex items-center gap-1.5 px-1.5 py-0.5 rounded border font-semibold transition-colors ${tone}`}
    86→      title={`threat condition L${level} — ${label}${quarantined ? ` · ${quarantined} route(s) quarantined` : anomalous ? ` · ${anomalous} anomalous` : ''}`}
    87→    >
    88→      <span className="flex gap-[2px]" aria-hidden>
    89→        {[1, 2, 3, 4, 5].map((i) => (
    90→          <span key={i} className={`w-[3px] h-2 rounded-[1px] ${i >= level ? 'bg-current' : 'bg-slate-700/60'}`} />
    91→        ))}
    92→      </span>
    93→      <span className="text-[9.5px] tracking-wider">L{level} {label}</span>
    94→    </span>
    95→  );
    96→}

   132→/** Attack lifecycle progress steps: injected → detected → mitigated → rolledback */
   133→function PhaseSteps({ phase }: { phase: RunResult['phase'] }) {
   134→  const steps: { id: RunResult['phase']; label: string }[] = [
   135→    { id: 'injected', label: 'injected' },
   136→    { id: 'detected', label: 'detected' },
   137→    { id: 'mitigated', label: 'mitigated' },
   138→    { id: 'rolledback', label: 'rolled back' },
   139→  ];
   140→  const order: Record<string, number> = { idle: 0, injected: 1, detected: 2, mitigated: 3, rolledback: 4, failed: 4 };
   141→  const current = order[phase] ?? 0;
   142→  return (
   143→    <div className="flex items-center gap-1">
   144→      {steps.map((s, i) => {
   145→        const reached = current > i + 1 || (current === i + 1);
   146→        const isFailed = phase === 'failed' && i === steps.length - 1;
   147→        const isCurrent = current === i + 1 && phase !== 'failed';
   148→        return (
   149→          <React.Fragment key={s.id}>
   150→            {i > 0 && <span className={`text-[9px] ${reached ? 'text-slate-500' : 'text-slate-700'}`}>──</span>}
   151→            <span
   152→              className={`text-[9px] font-mono px-1.5 py-0.5 rounded border ${
   153→                isFailed
   154→                  ? 'border-red-900 text-red-400 bg-red-950/40'
   155→                  : reached
   156→                    ? isCurrent
   157→                      ? 'border-amber-600 text-amber-300 bg-amber-950/50 animate-pulse'
   158→                      : 'border-emerald-800 text-emerald-300 bg-emerald-950/40'
   159→                    : 'border-slate-800 text-slate-600'
   160→              }`}
   161→            >
   162→              {s.label}
   163→            </span>
   164→          </React.Fragment>
   165→        );
   166→      })}
   167→    </div>
   168→  );
   169→}
