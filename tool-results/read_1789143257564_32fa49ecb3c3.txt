    33→/**
    34→ * NOC threat-condition level (DEFCON-style): derives posture from routes +
    35→ * active run phase. L5 all-clear → L1 active hijack quarantine.
    36→ */
    37→function ThreatCondition({ state }: { state: SimState }) {
    38→  const routes = Object.values(state.routes);
    39→  const quarantined = routes.filter((r) => r.underOverride && r.route.locPref === 0).length;
    40→  const anomalous = routes.filter((r) => r.status !== 'normal' || r.underOverride).length;
    41→  const phase = state.activeRun?.phase;
    42→
    43→  let level = 5;
    44→  let label = 'ALL CLEAR';
    45→  let tone = 'border-emerald-800 text-emerald-300 bg-emerald-950/40';
    46→  if (quarantined > 0 || phase === 'mitigated') {
    47→    level = 1;
    48→    label = 'QUARANTINE';
    49→    tone = 'border-red-700 text-red-300 bg-red-950/50';
    50→  } else if (phase === 'detected' || phase === 'injected') {
    51→    level = 2;
    52→    label = 'ENGAGED';
    53→    tone = 'border-amber-700 text-amber-300 bg-amber-950/40';
    54→  } else if (anomalous > 0) {
    55→    level = 3;
    56→    label = 'ELEVATED';
    57→    tone = 'border-orange-800 text-orange-300 bg-orange-950/40';
    58→  } else if (state.metrics.totalRuns > 0 && state.metrics.msrPercent < 100) {
    59→    level = 4;
    60→    label = 'GUARDED';
    61→    tone = 'border-yellow-800 text-yellow-300 bg-yellow-950/30';
    62→  }
    63→
    64→  return (
    65→    <span
    66→      className={`hidden sm:inline-flex items-center gap-1.5 px-1.5 py-0.5 rounded border font-semibold transition-colors ${tone}`}
    67→      title={`threat condition L${level} — ${label}${quarantined ? ` · ${quarantined} route(s) quarantined` : anomalous ? ` · ${anomalous} anomalous` : ''}`}
    68→    >
    69→      <span className="flex gap-[2px]" aria-hidden>
    70→        {[1, 2, 3, 4, 5].map((i) => (
    71→          <span key={i} className={`w-[3px] h-2 rounded-[1px] ${i >= level ? 'bg-current' : 'bg-slate-700/60'}`} />
    72→        ))}
    73→      </span>
    74→      <span className="text-[9.5px] tracking-wider">L{level} {label}</span>
    75→    </span>
    76→  );
    77→}

   113→/** Attack lifecycle progress steps: injected → detected → mitigated → rolledback */
   114→function PhaseSteps({ phase }: { phase: RunResult['phase'] }) {
   115→  const steps: { id: RunResult['phase']; label: string }[] = [
   116→    { id: 'injected', label: 'injected' },
   117→    { id: 'detected', label: 'detected' },
   118→    { id: 'mitigated', label: 'mitigated' },
   119→    { id: 'rolledback', label: 'rolled back' },
   120→  ];
   121→  const order: Record<string, number> = { idle: 0, injected: 1, detected: 2, mitigated: 3, rolledback: 4, failed: 4 };
   122→  const current = order[phase] ?? 0;
   123→  return (
   124→    <div className="flex items-center gap-1">
   125→      {steps.map((s, i) => {
   126→        const reached = current > i + 1 || (current === i + 1);
   127→        const isFailed = phase === 'failed' && i === steps.length - 1;
   128→        const isCurrent = current === i + 1 && phase !== 'failed';
   129→        return (
   130→          <React.Fragment key={s.id}>
   131→            {i > 0 && <span className={`text-[9px] ${reached ? 'text-slate-500' : 'text-slate-700'}`}>──</span>}
   132→            <span
   133→              className={`text-[9px] font-mono px-1.5 py-0.5 rounded border ${
   134→                isFailed
   135→                  ? 'border-red-900 text-red-400 bg-red-950/40'
   136→                  : reached
   137→                    ? isCurrent
   138→                      ? 'border-amber-600 text-amber-300 bg-amber-950/50 animate-pulse'
   139→                      : 'border-emerald-800 text-emerald-300 bg-emerald-950/40'
   140→                    : 'border-slate-800 text-slate-600'
   141→              }`}
   142→            >
   143→              {s.label}
   144→            </span>
   145→          </React.Fragment>
   146→        );
   147→      })}
   148→    </div>
   149→  );
   150→}
