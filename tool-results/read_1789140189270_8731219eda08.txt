    30→/**
    31→ * NOC threat-condition level (DEFCON-style): derives posture from routes +
    32→ * active run phase. L5 all-clear → L1 active hijack quarantine.
    33→ */
    34→function ThreatCondition({ state }: { state: SimState }) {
    35→  const routes = Object.values(state.routes);
    36→  const quarantined = routes.filter((r) => r.underOverride && r.route.locPref === 0).length;
    37→  const anomalous = routes.filter((r) => r.status !== 'normal' || r.underOverride).length;
    38→  const phase = state.activeRun?.phase;
    39→
    40→  let level = 5;
    41→  let label = 'ALL CLEAR';
    42→  let tone = 'border-emerald-800 text-emerald-300 bg-emerald-950/40';
    43→  if (quarantined > 0 || phase === 'mitigated') {
    44→    level = 1;
    45→    label = 'QUARANTINE';
    46→    tone = 'border-red-700 text-red-300 bg-red-950/50';
    47→  } else if (phase === 'detected' || phase === 'injected') {
    48→    level = 2;
    49→    label = 'ENGAGED';
    50→    tone = 'border-amber-700 text-amber-300 bg-amber-950/40';
    51→  } else if (anomalous > 0) {
    52→    level = 3;
    53→    label = 'ELEVATED';
    54→    tone = 'border-orange-800 text-orange-300 bg-orange-950/40';
    55→  } else if (state.metrics.totalRuns > 0 && state.metrics.msrPercent < 100) {
    56→    level = 4;
    57→    label = 'GUARDED';
    58→    tone = 'border-yellow-800 text-yellow-300 bg-yellow-950/30';
    59→  }
    60→
    61→  return (
    62→    <span
    63→      className={`hidden sm:inline-flex items-center gap-1.5 px-1.5 py-0.5 rounded border font-semibold transition-colors ${tone}`}
    64→      title={`threat condition L${level} — ${label}${quarantined ? ` · ${quarantined} route(s) quarantined` : anomalous ? ` · ${anomalous} anomalous` : ''}`}
    65→    >
    66→      <span className="flex gap-[2px]" aria-hidden>
    67→        {[1, 2, 3, 4, 5].map((i) => (
    68→          <span key={i} className={`w-[3px] h-2 rounded-[1px] ${i >= level ? 'bg-current' : 'bg-slate-700/60'}`} />
    69→        ))}
    70→      </span>
    71→      <span className="text-[9.5px] tracking-wider">L{level} {label}</span>
    72→    </span>
    73→  );
    74→}

   110→/** Attack lifecycle progress steps: injected → detected → mitigated → rolledback */
   111→function PhaseSteps({ phase }: { phase: RunResult['phase'] }) {
   112→  const steps: { id: RunResult['phase']; label: string }[] = [
   113→    { id: 'injected', label: 'injected' },
   114→    { id: 'detected', label: 'detected' },
   115→    { id: 'mitigated', label: 'mitigated' },
   116→    { id: 'rolledback', label: 'rolled back' },
   117→  ];
   118→  const order: Record<string, number> = { idle: 0, injected: 1, detected: 2, mitigated: 3, rolledback: 4, failed: 4 };
   119→  const current = order[phase] ?? 0;
   120→  return (
   121→    <div className="flex items-center gap-1">
   122→      {steps.map((s, i) => {
   123→        const reached = current > i + 1 || (current === i + 1);
   124→        const isFailed = phase === 'failed' && i === steps.length - 1;
   125→        const isCurrent = current === i + 1 && phase !== 'failed';
   126→        return (
   127→          <React.Fragment key={s.id}>
   128→            {i > 0 && <span className={`text-[9px] ${reached ? 'text-slate-500' : 'text-slate-700'}`}>──</span>}
   129→            <span
   130→              className={`text-[9px] font-mono px-1.5 py-0.5 rounded border ${
   131→                isFailed
   132→                  ? 'border-red-900 text-red-400 bg-red-950/40'
   133→                  : reached
   134→                    ? isCurrent
   135→                      ? 'border-amber-600 text-amber-300 bg-amber-950/50 animate-pulse'
   136→                      : 'border-emerald-800 text-emerald-300 bg-emerald-950/40'
   137→                    : 'border-slate-800 text-slate-600'
   138→              }`}
   139→            >
   140→              {s.label}
   141→            </span>
   142→          </React.Fragment>
   143→        );
   144→      })}
   145→    </div>
   146→  );
   147→}
